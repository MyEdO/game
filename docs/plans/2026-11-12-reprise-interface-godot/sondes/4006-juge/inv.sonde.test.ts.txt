function piloter(max = 80): string {
  for (let i = 0; i < max; i++) {
    const g = get();
    if (g.pendingRest) return 'halte';
    if (g.travelRecap || !g.travelPlan) return 'arrivee/fin';
    if (g.battle) return 'combat';
    if (g.pendingTest) { if (g.pendingTest.roll == null) g.jetLancer('test'); get().jetValider('test'); }
    else if (g.pendingCascade) avanceEtapeCascade(get);
    else return 'MORT(aucun slot, pas de halte)';
    banc.reprendreTout();
  }
  return 'boucle';
}
