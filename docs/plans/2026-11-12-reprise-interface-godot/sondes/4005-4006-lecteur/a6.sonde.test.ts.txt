const trace: string[] = [];
const etat = (get: any) => `slot=${get().pendingCascade?.purpose ?? 'null'}[${get().pendingCascade?.title ?? ''}] cursor=${get().pendingCascade?.cursor}/${get().pendingCascade?.participants?.length} pile=[${get().suspendedCascades.map((p: any) => p.purpose + ':' + p.title + '@' + p.cursor + '/' + p.participants.length).join(',')}] pendingTest=${!!get().pendingTest} travelRecap=${!!get().travelRecap}`;
const orig = (TABLES.appliers as any).etapeRencontre;
let INJECTER = true;
let MODE: 'choice' | 'volontaires' = 'choice';
let storeGet: any = null;
(TABLES.appliers as any).etapeRencontre = (get: any, set: any, step: any, hero: any) => {
  const r = orig(get, set, step, hero);
  storeGet = get;
  if (!INJECTER) { trace.push(`[applier etapeRencontre] TEMOIN sans runFlow tiree=${step.table?.result?.id} ${etat(get)}`); return r; }
  trace.push(`[applier etapeRencontre] tiree=${step.table?.result?.id} AVANT runFlow ${etat(get)}`);
  const fx = { kind: 'do', effect: { type: 'ops', on: 'party', ops: [{ op: 'condition', id: { id: 'extenue' }, value: 1 }] } };
  const flow = MODE === 'choice' ? { kind: 'choice', prompt: 'Presser le pas ?', oui: fx }
    : { kind: 'do', effect: { type: 'ops', on: { volontaires: 'Presser le pas' }, ops: [{ op: 'condition', id: { id: 'extenue' }, value: 1 }] } };
  jouerFlowEntier(runFlow(get, set, flow as any, 'Voyage fatigant !'));
  trace.push(`[applier etapeRencontre] APRÈS runFlow ${etat(get)}`);
  return r;
};
