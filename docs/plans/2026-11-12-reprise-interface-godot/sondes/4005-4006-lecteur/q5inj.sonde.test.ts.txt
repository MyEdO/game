function jouer(titre: string, id: string, graine: number, preparer?: () => void) {
  freshState(graine);
  preparer?.();
  out.push(`\n===== ${titre} (graine ${graine})`);
  const plan = buildSeaPlan(get as any, 'r1', 'A', 'B', seaMap.routes[0], { cadence: 'jourParJour' })!;
  set({ travelPlan: { ...plan, sea: { ...plan.sea!, daysToEvent: 1, forcedEventId: undefined } } } as never);
  runSeaDay(get as any, set as any);
  out.push(`apres runSeaDay : ${etat()}`);
  for (let i = 0; i < 40; i++) {
    const g: any = get();
    let acte = '';
    if (g.pendingTest) { if (g.pendingTest.roll == null) g.jetLancer('test'); get().jetValider('test'); acte = 'jetValider(test)'; }
    else if (g.pendingCascade) { acte = 'avanceEtape(' + (avanceEtapeCascade(get as any) ?? '?') + ')'; }
    else { out.push(`pas ${i} : AUCUN slot ni pendingTest -> stop`); break; }
    banc.reprendreTout();
    out.push(`pas ${i} ${acte} -> ${etat()}`);
    if (get().pendingRest && !(get() as any).__vu) { out.push('>>> pendingRest ouvert (on continue de jouer le slot restant)'); (get() as any).__vu = true; }
  }
  out.push(`FINAL ${etat()}`);
  out.push(`journal(queue): ${JSON.stringify(get().journal.slice(-6))}`);
}
