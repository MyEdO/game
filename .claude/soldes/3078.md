from collections import Counter
import json, re, subprocess, unicodedata

sha = '6940114e654b311de9a42d649fa9c6da0263a197'
def lire(nom):
    return json.loads(subprocess.check_output(['git', 'show', f'{sha}:src/data/{nom}.json']))

categories = {'melee', 'distance', 'ammunition'}
entrees = [x for x in lire('trappings') if x['categorie'] in categories]
catalogue = {x['id']: x['categorie'] for x in entrees}

def termes(texte):
    texte = unicodedata.normalize('NFKD', texte.lower().replace('œ', 'oe').replace('æ', 'ae'))
    texte = ''.join(c for c in texte if not unicodedata.combining(c))
    return ' '.join(re.sub('[sx]$', '', re.sub('aux$', 'al', mot)) for mot in re.findall('[a-z0-9]+', texte))

lexique = {(x['id'], termes(nom)) for x in entrees for nom in (x['id'], x['label'])}

def feuilles(refs, chemin=()):
    for i, ref in enumerate(refs):
        site = (*chemin, i)
        if 'of' in ref:
            yield from feuilles(ref['of'], (*site, 'of'))
        elif 'lot' in ref:
            yield from feuilles(ref['lot'], (*site, 'lot'))
        else:
            yield site, ref

for nom in ('careerLevels', 'classes', 'creatures', 'pregens'):
    porteurs = lire(nom)
    compte = Counter()
    for porteur in porteurs:
        compte['racines'] += len(porteur['trappings'])
        for site, ref in feuilles(porteur['trappings']):
            compte['feuilles'] += 1
            categorie = catalogue.get(ref.get('id')) if 'type' not in ref else None
            if categorie:
                compte[categorie] += 1
                compte['imbriquees'] += 'of' in site
                compte['dans_lot'] += 'lot' in site
            if 'text' in ref or ('nature' in ref and 'label' in ref):
                forme = 'texte' if 'text' in ref else 'mention'
                compte[forme] += 1
                texte = ref.get('text', ref.get('label'))
                candidats = sorted({id for id, terme in lexique if terme and f' {terme} ' in f' {termes(texte)} '})
                compte['candidats'] += bool(candidats)
                print(nom, porteur['id'], site, forme, texte, candidats)
    print(nom, len(porteurs), dict(compte))
