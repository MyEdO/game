```python
import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import signal
import subprocess
import time

parser = argparse.ArgumentParser()
parser.add_argument("code", type=Path)
parser.add_argument("corpus", type=Path)
parser.add_argument("banc", type=Path)
parser.add_argument("mode", choices=("balayages", "perimetre"))
parser.add_argument("--memos", type=Path)
parser.add_argument("--base", default="ea4c97711b3135fe22d17accf962c10dfd39ef48")
parser.add_argument("--profil", action="store_true")
args = parser.parse_args()
args.banc.mkdir(parents=True)
memos = args.memos or args.banc / "memos"
entree = dict(racine=str(args.corpus), dossierDesMemos=str(memos), mode=args.mode)
if args.mode == "perimetre":
    entree["base"] = args.base
(args.banc / "entree.json").write_text(json.dumps(entree))
quota, period = Path("/sys/fs/cgroup/cpu.max").read_text().split()
capacite = int(quota) / int(period)

def cpu():
    valeurs = dict(ligne.split() for ligne in Path("/sys/fs/cgroup/cpu.stat").read_text().splitlines())
    return int(valeurs["usage_usec"])

cmd = ["node", "--max-old-space-size=1024", "--trace-gc-nvp"]
if args.profil:
    cmd += ["--cpu-prof", f"--cpu-prof-dir={args.banc}"]
cmd += [str(args.code / "scripts/test/perimetreDerivation.mjs")]
cmd += [str(args.banc / nom) for nom in ("entree.json", "sortie.json", "stade.json")]
with (args.banc / "node.log").open("w") as journal:
    debut, cpu0 = time.monotonic(), cpu()
    enfant = subprocess.Popen(cmd, cwd=args.corpus, stdout=journal, stderr=subprocess.STDOUT, start_new_session=True)
    dernier, dernier_cpu = debut, cpu0
    echantillons = []
    delai = False
    while True:
        pid, statut, res = os.wait4(enfant.pid, os.WNOHANG)
        maintenant = time.monotonic()
        if maintenant - dernier >= 1:
            usage = cpu()
            echantillons.append((usage - dernier_cpu) / (maintenant - dernier) / 10000 / capacite)
            dernier, dernier_cpu = maintenant, usage
        if pid:
            enfant.returncode = os.waitstatus_to_exitcode(statut)
            break
        if maintenant - debut >= 180 and not delai:
            delai = True
            os.killpg(enfant.pid, signal.SIGKILL)
        time.sleep(.1)
temps = maintenant - debut
mio = 1024**2
journal = (args.banc / "node.log").read_text(errors="replace")
tailles = list(map(int, re.findall(r"\bstart_object_size=(\d+)", journal)))
stade = args.banc / "stade.json"
sortie = args.banc / "sortie.json"
mesure = dict(temps_s=round(temps, 3), code=enfant.returncode, delai=delai,
    charge_pct=[round(x, 1) for x in (min(echantillons, default=0), (cpu()-cpu0)/temps/10000/capacite, max(echantillons, default=0))],
    cpu_processus_s=round(res.ru_utime+res.ru_stime, 3),
    rss_mio=round(res.ru_maxrss/1024, 1), objets_mio=round(max(tailles, default=0)/mio, 1),
    memos=[ligne for ligne in journal.splitlines() if ligne.startswith("[mémo]")])
if stade.exists():
    mesure["stade"] = json.loads(stade.read_bytes())
if sortie.exists():
    octets = sortie.read_bytes()
    dto = json.loads(octets)
    mesure["dto"] = dict(sha256=hashlib.sha256(octets).hexdigest(), post=len(dto["images"]["post"]),
        touches=len(dto["images"]["touches"]), modules=dto["modules"], retenus=len(dto["retenus"]),
        vitest=len(dto["vitest"]), node=len(dto["node"]), nonResolus=len(dto["nonResolus"]))
(args.banc / "mesure.json").write_text(json.dumps(mesure, ensure_ascii=False, indent=2))
print(json.dumps(mesure, ensure_ascii=False))
raise SystemExit(124 if delai else enfant.returncode)
```
