## Exhaustivité des defs

La sonde rapproche les fichiers déclarés sur disque et les registres chargés. Elle inspecte l’argument de déclaration dans l’AST, compare l’inventaire statique des niches aux expositions réelles, puis compare les clés et leur ordre def par def. Elle vérifie le rattachement au fichier et compare le jeu des datasets miscast aux clés de leur def. Les sept collections sans `{ id, label }` correspondent à `HORS_DE_LA_PORTE` dans `scripts/gen-espaces.mts`. Elles conservent leur route et leur clé Codex.

Écrire cette sonde dans `.git/sonde-3923.mts` dans la copie de preuve après `npm run gen`, puis exécuter `node --import tsx .git/sonde-3923.mts`.

```typescript
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { lireDefs } from '../scripts/gen-registry.mjs';
import { analyserCorpus, typescript } from '../scripts/guards/lib/dialecte.mjs';
import { SCHEMA_DEFS } from '../src/data/schemas/_registry.generated';
import { SCHEMA_DEFS_SCENES } from '../src/data/schemas/_registry-scenes.generated';
import { CATEGORY_DATASET_DERIVE, DATASET_FICHIER_DERIVE } from '../src/data/schemas/exposition-derivee';
import { CLES_CODEX, DATASET_DE_LA_CATEGORIE } from '../src/data/schemas/_cles-de-dataset.generated';
import { MISCAST_DATASETS } from '../src/engine/miscast';

const defs = [...SCHEMA_DEFS, ...SCHEMA_DEFS_SCENES];
const niches = [];
const ts = typescript();
const files = ['defs', 'defs-scenes'].flatMap(dir => {
  const root = 'src/data/schemas/' + dir;
  return lireDefs(root, ['file']).filter(def => def.file !== undefined).map(({ module, file }) => {
    const rel = root + '/' + module;
    return { rel, file, text: readFileSync(rel, 'utf8') };
  });
});
assert.deepEqual(files.map(def => def.file).sort(), defs.map(def => def.file).sort());
for (const { fichier: { rel }, sourceFile: source, diagnostics } of analyserCorpus(files)) {
  assert.deepEqual(diagnostics, [], rel);
  const property = (node, key) => node.properties.find(prop => prop.name?.getText(source) === key);
  const visit = node => {
    if (ts.isCallExpression(node) && node.expression.getText(source) === 'document') {
      const declaration = node.arguments[4];
      if (declaration && ts.isObjectLiteralExpression(declaration)) {
        const route = property(declaration, 'edit')?.initializer;
        if (route && ts.isObjectLiteralExpression(route) && property(route, 'niche')) {
          assert.equal(property(declaration, 'codex'), undefined, rel);
          for (const other of ['dataset', 'none', 'object']) assert.equal(property(route, other), undefined, rel);
          const file = source.statements.flatMap(s => ts.isVariableStatement(s) ? [...s.declarationList.declarations] : [])
            .find(v => v.name.getText(source) === 'file')?.initializer;
          assert(file && ts.isStringLiteral(file), rel);
          niches.push(file.text);
        }
      }
    }
    node.forEachChild(visit);
  };
  visit(source);
}

const live = defs.filter(def => def.exposition && 'niche' in def.exposition.edit);
assert.equal(new Set(niches).size, niches.length);
assert.deepEqual(niches.sort(), live.map(def => def.file).sort());
let categories = 0;
const sansEntrees = [];
for (const { file, exposition: { codex, edit } } of live) {
  const keys = Object.keys(edit.niche.categories);
  assert(keys.length > 0, file);
  assert.deepEqual(codex.keys, keys, file);
  for (const key of keys) {
    assert(CLES_CODEX.includes(key), key);
    assert.equal(DATASET_FICHIER_DERIVE[CATEGORY_DATASET_DERIVE[key]], file, key);
    const route = DATASET_DE_LA_CATEGORIE[key];
    if (route) assert.equal(route.fichier, file, key);
    else sansEntrees.push(key);
  }
  categories += keys.length;
}
const miscast = live.find(def => def.file === 'miscast.json');
assert(miscast);
assert.deepEqual([...MISCAST_DATASETS].sort(), Object.keys(miscast.exposition.edit.niche.categories).sort());
console.log(`${defs.length} defs (${SCHEMA_DEFS.length} données + ${SCHEMA_DEFS_SCENES.length} scènes) · ${live.length} niches · ${categories} catégories · 0 codex explicite · 0 route mixte · 0 écart de clés ou de fichier`);
console.log(niches.join(', '));
console.log('Niches hors lookup { id, label } : ' + sansEntrees.sort().join(', '));
console.log('MISCAST_DATASETS (' + MISCAST_DATASETS.length + ') : ' + MISCAST_DATASETS.join(', '));
```

## Refus à la compilation

```bash
]:
    path = Path(f'src/data/schemas/defs/{name}.ts')
```
