```ts
import { createHash } from 'node:crypto';
import { writeFileSync } from 'node:fs';
import { Session } from 'node:inspector/promises';
import { expect, it } from 'vitest';
import {
  auditerLEditabilite, orphanFields, programmeDuPerimetre, RACINES,
} from '../scripts/guards/lib/editabiliteDesChamps.mjs';
import { ecrituresDeLaCarte, ecrituresDuNarratif, ecrituresJouees } from './ecrituresJouees.testkit';

const banc = { narratif: ecrituresDuNarratif, carte: ecrituresDeLaCarte, scene: ecrituresJouees };
const racines = (process.env.RACINES_4608 ?? 'narratif').split(',') as (keyof typeof banc)[];
const sortie = process.env.SORTIE_4608!;
for (const racine of racines) it(racine, { timeout: 245_000 }, async () => {
  const mesurer = <T,>(phase: string, jouer: () => T): T => {
    const debut = performance.now();
    const resultat = jouer();
    console.error(racine, phase, (performance.now() - debut).toFixed(1), 'ms');
    return resultat;
  };
  const debut = performance.now();
  const programme = mesurer('programmeDuPerimetre', () => programmeDuPerimetre(process.cwd(), Object.values(RACINES)));
  try {
    const cpu = process.env.CPU_4608 === '1' ? new Session() : undefined;
    if (cpu) {
      cpu.connect();
      await cpu.post('Profiler.enable');
      await cpu.post('Profiler.setSamplingInterval', { interval: 2000 });
      await cpu.post('Profiler.start');
    }
    const ecritures = mesurer('ecrituresJouees', banc[racine]);
    const profil = cpu && (await cpu.post('Profiler.stop')).profile;
    cpu?.disconnect();
    const audit = mesurer('auditerLEditabilite', () => auditerLEditabilite(
      process.cwd(), programme, { racine: RACINES[racine], ecrituresJouees: ecritures },
    ));
    console.error(racine, 'total', (performance.now() - debut).toFixed(1), 'ms');
    if (profil) writeFileSync(`${sortie}/ecritures.cpuprofile`, JSON.stringify(profil));
    writeFileSync(`${sortie}/${racine}.json`, JSON.stringify(audit) + '\n');
    const sha = createHash('sha256');
    for (const ecriture of ecritures) sha.update(JSON.stringify(ecriture)).update('\n');
    console.error(racine, { ecritures: ecritures.length, sha: sha.digest('hex'), champs: audit.length });
    expect(orphanFields(audit)).toEqual([]);
    expect(audit.length).toBeGreaterThanOrEqual({ narratif: 360, carte: 500, scene: 415 }[racine]);
  } finally {
    programme.dispose();
  }
});
```
